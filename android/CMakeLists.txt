cmake_minimum_required(VERSION 3.22)
project(dread_android LANGUAGES C CXX)
option(SWR_ANDROID_VULKAN "Link the experimental native Vulkan renderer" OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(RT ${CMAKE_CURRENT_LIST_DIR}/../runtime)
set(SOURCES plat mem kernel ipc svc_core svc_hid svc_fs svc_vi nvdrv gpu gpu_engines
    gpu_3d loader main audio svc_launcher gpu_gl gl_load shader astc glq glq_gen
    dlss_gl overlay lsfg_stub host1x snxvk_glue backend_android android_driver)
list(TRANSFORM SOURCES PREPEND "${RT}/")
list(TRANSFORM SOURCES APPEND ".c")
add_library(dread SHARED ${SOURCES} ${RT}/bcn/bcn.cpp ${RT}/bcn/bcdec.cpp ${RT}/dread_ui/stub.c)
target_include_directories(dread PRIVATE ${RT} ${RT}/bcn ${RT}/oh264/include
    ${RT}/vp9/include ${CMAKE_CURRENT_LIST_DIR}/headers)
target_compile_definitions(dread PRIVATE SWR_RELEASE)
target_compile_options(dread PRIVATE -O2 -fno-strict-aliasing -fwrapv -Wno-format)
target_link_options(dread PRIVATE -Wl,-z,max-page-size=16384 -Wl,--no-undefined)
target_link_libraries(dread PRIVATE android log EGL GLESv3 aaudio z dl m)
if(SWR_ANDROID_VULKAN)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
    add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/local/libadrenotools adrenotools)
    foreach(hook main_hook hook_impl file_redirect_hook gsl_alloc_hook)
        target_link_options(${hook} PRIVATE -Wl,-z,max-page-size=16384)
    endforeach()
    target_link_libraries(dread PRIVATE adrenotools)
    target_compile_definitions(dread PRIVATE SWR_HAVE_SNXVK VK_USE_PLATFORM_ANDROID_KHR VK_NO_PROTOTYPES)
    target_include_directories(dread PRIVATE ${RT}/snxvk/deps/Vulkan-Headers-1.4.365/include)
    target_link_libraries(dread PRIVATE ${CMAKE_CURRENT_LIST_DIR}/local/vulkan/libsnxvk.a)
endif()
set(DEPS ${CMAKE_CURRENT_LIST_DIR}/local)
if(EXISTS ${DEPS}/libvpx/libvpx.a)
    target_compile_definitions(dread PRIVATE SWR_HAVE_VP9)
    target_link_libraries(dread PRIVATE ${DEPS}/libvpx/libvpx.a)
endif()
if(EXISTS ${DEPS}/openh264/libdecoder.a)
    target_link_libraries(dread PRIVATE ${DEPS}/openh264/libdecoder.a ${DEPS}/openh264/libcommon.a)
else()
    target_compile_definitions(dread PRIVATE SWR_NO_H264)
endif()
if(GAME_BUILD_DIR)
    file(GLOB GAME_SOURCES CONFIGURE_DEPENDS "${GAME_BUILD_DIR}/*.c")
    add_library(MetroidDread_code SHARED ${GAME_SOURCES})
    target_include_directories(MetroidDread_code PRIVATE ${RT})
    target_compile_options(MetroidDread_code PRIVATE -O2 -fno-strict-aliasing -fwrapv
        -fvisibility=hidden -w -include ${CMAKE_CURRENT_LIST_DIR}/compat.h)
    target_link_options(MetroidDread_code PRIVATE -Wl,-z,max-page-size=16384 -Wl,--no-undefined)
    target_link_libraries(MetroidDread_code PRIVATE dl m)
endif()
